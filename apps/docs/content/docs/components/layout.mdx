## Usage

```tsx
import { Layout } from 'michael-ui';

export default () => {
  return <Layout />;
};
```



## API Reference

```jsx
<Layout>
  <Header>header</Header>
  <Layout>
    <Sider>left sidebar</Sider>
    <Content>main content</Content>
    <Sider>right sidebar</Sider>
  </Layout>
  <Footer>footer</Footer>
</Layout>
```
